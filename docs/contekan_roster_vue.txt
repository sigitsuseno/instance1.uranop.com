// ─── View State ───────────────────────────────────
const viewMode = ref('table')
const selectedDepartment = ref('')
const currentYear = ref(new Date().getFullYear())
const currentMonth = ref(new Date().getMonth() + 1)

const monthNames = ['', 'Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni',
  'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember']
const dayHeaders = ['Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu', 'Minggu']

const periodStart = computed(() => {
  const d = new Date(currentYear.value, currentMonth.value - 1, 25)
  d.setMonth(d.getMonth() - 1)
  return `${d.getDate()} ${monthNames[d.getMonth() + 1]} ${d.getFullYear()}`

  const d = new Date(currentYear.value, currentMonth.value - 1, 24)
  return `${d.getDate()} ${monthNames[d.getMonth() + 1]} ${d.getFullYear()}`
})

const periodDates = computed(() => {
  const start = new Date(currentYear.value, currentMonth.value - 1, 25)
  start.setMonth(start.getMonth() - 1)
  const end = new Date(currentYear.value, currentMonth.value - 1, 24)
  const dates = []
  const cur = new Date(start)
  while (cur <= end) {
    const y = cur.getFullYear()
    const m = String(cur.getMonth() + 1).padStart(2, '0')
    const d = String(cur.getDate()).padStart(2, '0')
    dates.push(`${y}-${m}-${d}`)
    cur.setDate(cur.getDate() + 1)
  }
  return dates
})

function dayNumber(dateStr) {
  return parseInt(dateStr.split('-')[2])
}

function dayAbbr(dateStr) {
  const d = new Date(dateStr)
  const days = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab']
  return days[d.getDay()]
}

function isSunday(dateStr) {
  return new Date(dateStr).getDay() === 0

  return new Date(dateStr).getDay() === 6
}

// Calendar grid — summarise by day
const calendarGrid = computed(() => {
  const cells = []
  const firstDay = new Date(currentYear.value, currentMonth.value - 1, 1).getDay()
  const adjustedFirstDay = firstDay === 0 ? 6 : firstDay - 1
  const totalDays = new Date(currentYear.value, currentMonth.value, 0).getDate()
  const prevMonthDays = new Date(currentYear.value, currentMonth.value - 1, 0).getDate()

  // Previous month padding
  for (let i = adjustedFirstDay - 1; i >= 0; i--) {
    cells.push({ date: prevMonthDays - i, inMonth: false, workingCount: 0, offCount: 0 })
  }

  // Current month days
  for (let d = 1; d <= totalDays; d++) {
    const dateStr = `${currentYear.value}-${String(currentMonth.value).padStart(2, '0')}-${String(d).padStart(2, '0')}`
    let workingCount = 0
    let offCount = 0
    let unsetCount = 0

    filteredRoster.value.forEach(emp => {
      const sched = emp.schedule.find(s => s && s.date === dateStr)
      if (!sched) { unsetCount++; return }
      if (sched.is_off) offCount++
      else workingCount++
    })

    cells.push({
      date: d,
      inMonth: true,
      dateStr,
      workingCount,
      offCount,
      unsetCount,
    })
  }

  // Next month padding
  const remaining = 7 - (cells.length % 7)
  if (remaining < 7) {
    for (let i = 1; i <= remaining; i++) {
      cells.push({ date: i, inMonth: false, workingCount: 0, offCount: 0 })
    }
  }

  return cells
})

  return `${shiftColor.bg} ${shiftColor.text}`

// ─── Shift Color Mapping ──────────────────────────
function getShiftColor(code) {
  const map = {
    P: { bg: 'bg-(--primary)/20', text: 'text-(--primary)' },
    S: { bg: 'bg-(--warning)/20', text: 'text-(--warning)' },
    M: { bg: 'bg-(--info)/20', text: 'text-(--info)' },
    L: { bg: 'bg-(--danger)/20', text: 'text-(--danger)' },
  }
  const upper = (code || '').toUpperCase()
  if (map[upper]) return map[upper]
  // Default: hash based
  const hash = code ? code.charCodeAt(0) % 6 : 0
  const defaults = [
    { bg: 'bg-(--primary)/15', text: 'text-(--primary)' },
    { bg: 'bg-(--warning)/15', text: 'text-(--warning)' },
    { bg: 'bg-(--info)/15', text: 'text-(--info)' },
    { bg: 'bg-purple-500/15', text: 'text-purple-500' },
    { bg: 'bg-pink-500/15', text: 'text-pink-500' },
    { bg: 'bg-teal-500/15', text: 'text-teal-500' },
  ]
  return defaults[hash]
}

function cellBgClass(schedule) {
  if (!schedule) return 'bg-(--bg-elevated)'
  if (schedule.is_off) return 'bg-(--danger)/20'
  const c = getShiftColor(schedule.external_code || '')
  return c.bg
}

function cellTitle(emp, schedule) {

  return `${emp.name}: ${schedule.name || schedule.external_code || '-'}`
}

// ─── Fetch Data ───────────────────────────────────
async function fetchAll() {
  loading.value = true
  error.value = ''

  try {
    const [deptRes, rosterRes, shiftRes] = await Promise.all([
      get('/api/organization/departments/options').catch(() => ({ data: [] })),
      get(`/api/schedule/roster?year=${currentYear.value}&month=${currentMonth.value}`),
      get('/api/schedule/shifts').catch(() => ({ data: [] })),
    ])

    departments.value = deptRes?.data || []
    rosterData.value = rosterRes?.data || []
    allShifts.value = shiftRes?.data || []
  } catch (err) {
    error.value = err.message || 'Gagal memuat data roster'
  } finally {
    loading.value = false
  }
}

function prevPeriod() {
  if (currentMonth.value === 1) {
    currentMonth.value = 12
    currentYear.value--
  } else {
    currentMonth.value--
  }
}

function nextPeriod() {
  if (currentMonth.value === 12) {
    currentMonth.value = 1
    currentYear.value++
  } else {
    currentMonth.value++
  }
}

  overrideShiftId.value = existing?.shift_id || null

// ─── Utils ────────────────────────────────────────
function formatBytes(bytes) {
  if (bytes < 1024) return bytes + ' B'
  if (bytes < 1048576) return (bytes / 1024).toFixed(1) + ' KB'
  return (bytes / 1048576).toFixed(1) + ' MB'
}
